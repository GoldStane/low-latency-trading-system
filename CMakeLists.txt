cmake_minimum_required(VERSION 3.20)
project(low_latency_trading_system LANGUAGES CXX)

add_library(trading_book cpp/order_book/order_book.cpp)
target_include_directories(trading_book PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
target_compile_features(trading_book PUBLIC cxx_std_20)
set_target_properties(trading_book PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(trading_book PRIVATE /W4)
else()
    target_compile_options(trading_book PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(ENABLE_SANITIZERS "Enable ASan and UBSan (Clang/GCC)" OFF)
if(ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "Sanitizers require Clang or GCC")
    endif()
    target_compile_options(trading_book PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(trading_book PUBLIC -fsanitize=address,undefined)
endif()

add_library(trading_market_data cpp/market_data/normalized_csv_reader.cpp)
target_include_directories(trading_market_data PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
target_compile_features(trading_market_data PUBLIC cxx_std_20)
if(ENABLE_SANITIZERS)
    target_compile_options(trading_market_data PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(trading_market_data PUBLIC -fsanitize=address,undefined)
endif()

add_library(trading_replay cpp/replay/historical_replay.cpp)
target_link_libraries(trading_replay PUBLIC trading_market_data trading_book)
add_executable(historical_replay cpp/replay/main.cpp)
target_link_libraries(historical_replay PRIVATE trading_replay)
foreach(target trading_market_data trading_replay historical_replay)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

option(BUILD_BENCHMARKS "Build standalone order book baseline benchmark" OFF)
if(BUILD_BENCHMARKS)
    add_executable(order_book_benchmark benchmarks/order_book_benchmark.cpp)
    target_link_libraries(order_book_benchmark PRIVATE trading_book)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(normalized_csv_tests tests/unit/test_normalized_csv.cpp)
    target_link_libraries(normalized_csv_tests PRIVATE trading_market_data)
    add_test(NAME normalized_csv COMMAND normalized_csv_tests)
    add_executable(historical_replay_tests tests/integration/test_historical_replay.cpp)
    target_link_libraries(historical_replay_tests PRIVATE trading_replay)
    add_test(NAME historical_replay_pipeline COMMAND historical_replay_tests)
    add_test(NAME historical_replay_cli
        COMMAND "${CMAKE_COMMAND}" -DREPLAY=$<TARGET_FILE:historical_replay>
            -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_replay_cli.cmake")
    add_executable(order_book_tests tests/unit/test_order_book.cpp)
    target_link_libraries(order_book_tests PRIVATE trading_book)
    add_test(NAME order_book COMMAND order_book_tests)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_test(NAME market_data_analysis
        COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests/unit -v)
    set_tests_properties(market_data_analysis PROPERTIES
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
