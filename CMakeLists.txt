cmake_minimum_required(VERSION 3.20)
project(low_latency_trading_system LANGUAGES CXX)

add_library(trading_book cpp/order_book/order_book.cpp)
target_include_directories(trading_book PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
target_compile_features(trading_book PUBLIC cxx_std_20)
set_target_properties(trading_book PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(trading_book PRIVATE /W4)
else()
    target_compile_options(trading_book PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(ENABLE_SANITIZERS "Enable ASan and UBSan (Clang/GCC)" OFF)
if(ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "Sanitizers require Clang or GCC")
    endif()
    target_compile_options(trading_book PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(trading_book PUBLIC -fsanitize=address,undefined)
endif()

add_library(trading_market_data
    cpp/market_data/normalized_csv_reader.cpp
    cpp/market_data/market_data_source.cpp
    cpp/market_data/normalized_csv_source.cpp
    cpp/market_data/ibkr/quote_translator.cpp)
add_library(trading_runtime_core cpp/runtime/order_book_market_data_handler.cpp)
target_link_libraries(trading_runtime_core PUBLIC trading_market_data trading_book)
target_include_directories(trading_market_data PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/cpp")
target_compile_features(trading_market_data PUBLIC cxx_std_20)
if(ENABLE_SANITIZERS)
    target_compile_options(trading_market_data PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(trading_market_data PUBLIC -fsanitize=address,undefined)
endif()

add_library(trading_replay cpp/replay/historical_replay.cpp)
target_link_libraries(trading_replay PUBLIC trading_runtime_core)
add_executable(historical_replay cpp/replay/main.cpp)
target_link_libraries(historical_replay PRIVATE trading_replay)
foreach(target trading_market_data trading_runtime_core trading_replay historical_replay)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

option(BUILD_BENCHMARKS "Build standalone benchmarks" OFF)
if(BUILD_BENCHMARKS)
    add_executable(event_pipeline_benchmark benchmarks/event_pipeline_benchmark.cpp)
    target_link_libraries(event_pipeline_benchmark PRIVATE trading_runtime_core)
    add_executable(order_book_benchmark benchmarks/order_book_benchmark.cpp)
    target_link_libraries(order_book_benchmark PRIVATE trading_book)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(event_pipeline_tests tests/integration/test_event_pipeline.cpp)
    target_link_libraries(event_pipeline_tests PRIVATE trading_replay)
    add_test(NAME event_pipeline COMMAND event_pipeline_tests)
    add_executable(ibkr_translation_tests tests/unit/test_ibkr_translation.cpp)
    target_link_libraries(ibkr_translation_tests PRIVATE trading_market_data)
    add_test(NAME ibkr_translation COMMAND ibkr_translation_tests)
    add_executable(normalized_csv_tests tests/unit/test_normalized_csv.cpp)
    target_link_libraries(normalized_csv_tests PRIVATE trading_market_data)
    add_test(NAME normalized_csv COMMAND normalized_csv_tests)
    add_executable(historical_replay_tests tests/integration/test_historical_replay.cpp)
    target_link_libraries(historical_replay_tests PRIVATE trading_replay)
    add_test(NAME historical_replay_pipeline COMMAND historical_replay_tests)
    add_test(NAME historical_replay_cli
        COMMAND "${CMAKE_COMMAND}" -DREPLAY=$<TARGET_FILE:historical_replay>
            -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_replay_cli.cmake")
    add_executable(order_book_tests tests/unit/test_order_book.cpp)
    target_link_libraries(order_book_tests PRIVATE trading_book)
    add_test(NAME order_book COMMAND order_book_tests)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_test(NAME market_data_analysis
        COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests/unit -v)
    set_tests_properties(market_data_analysis PROPERTIES
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()

option(ENABLE_IBKR "Build experimental IBKR market-data-only connectivity tool" OFF)
set(IBKR_API_ROOT "" CACHE PATH "External IBJts directory containing source/cppclient/client")
set(IBKR_API_LIBRARY "" CACHE FILEPATH "Externally built IBKR twsapi library (matching headers and architecture)")
set(IBKR_EXTRA_LIBRARIES "" CACHE STRING "Additional SDK dependencies, e.g. Intel decimal/Protobuf libraries")
if(ENABLE_IBKR)
    if(NOT UNIX)
        message(FATAL_ERROR "The experimental IBKR session currently supports Linux/macOS only.")
    endif()
    set(ibkr_include "${IBKR_API_ROOT}/source/cppclient/client")
    if(NOT EXISTS "${ibkr_include}/EClientSocket.h" OR NOT EXISTS "${ibkr_include}/DefaultEWrapper.h")
        message(FATAL_ERROR "ENABLE_IBKR requires the external TWS C++ SDK. Set IBKR_API_ROOT to IBJts (see docs/ibkr-market-data.md).")
    endif()
    if(NOT EXISTS "${IBKR_API_LIBRARY}")
        message(FATAL_ERROR "Set IBKR_API_LIBRARY to a compiled twsapi library matching your SDK headers and CPU; see docs/ibkr-market-data.md.")
    endif()
    find_package(Threads REQUIRED)
    find_package(Protobuf 3.12 REQUIRED)
    if(NOT EXISTS "${ibkr_include}/protobufUnix/TickPrice.pb.h")
        message(FATAL_ERROR "This adapter targets IBKR API 10.50.02 with generated protobufUnix headers.")
    endif()
    add_library(trading_ibkr_market_data cpp/market_data/ibkr/ibkr_source.cpp)
    target_include_directories(trading_ibkr_market_data SYSTEM PRIVATE "${ibkr_include}" "${ibkr_include}/protobufUnix")
    target_link_libraries(trading_ibkr_market_data PUBLIC trading_market_data PRIVATE
        "${IBKR_API_LIBRARY}" ${IBKR_EXTRA_LIBRARIES} Threads::Threads protobuf::libprotobuf)
    add_executable(ibkr_market_data cpp/market_data/ibkr/main.cpp)
    target_link_libraries(ibkr_market_data PRIVATE trading_ibkr_market_data)
    if(BUILD_TESTING)
        add_executable(ibkr_sdk_tests tests/unit/test_ibkr_sdk.cpp)
        target_include_directories(ibkr_sdk_tests SYSTEM PRIVATE "${ibkr_include}")
        target_link_libraries(ibkr_sdk_tests PRIVATE trading_ibkr_market_data)
        add_test(NAME ibkr_sdk_offline COMMAND ibkr_sdk_tests)
        add_test(NAME ibkr_cli COMMAND "${CMAKE_COMMAND}" -DIBKR=$<TARGET_FILE:ibkr_market_data>
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_ibkr_cli.cmake")
    endif()
    foreach(target trading_ibkr_market_data ibkr_market_data)
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()
endif()
