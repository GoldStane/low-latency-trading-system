# Optional standalone helper. Compiles external SDK sources without copying them.
cmake_minimum_required(VERSION 3.20)
project(external_ibkr_sdk LANGUAGES CXX)
set(IBKR_API_ROOT "" CACHE PATH "External IBJts SDK directory")
set(IBKR_DECIMAL_LIBRARY "" CACHE FILEPATH "Intel decimal library built for this CPU")
set(api "${IBKR_API_ROOT}/source/cppclient/client")
if(NOT EXISTS "${api}/EClientSocket.cpp" OR NOT EXISTS "${api}/protobufUnix/TickPrice.pb.cc")
    message(FATAL_ERROR "Set IBKR_API_ROOT to the external IBKR 10.50.02 SDK directory")
endif()
if(NOT EXISTS "${IBKR_DECIMAL_LIBRARY}")
    message(FATAL_ERROR "Set IBKR_DECIMAL_LIBRARY to your built libbid.a (see SDK Intel_lib_build.txt)")
endif()
find_package(Protobuf 3.12 REQUIRED)
find_package(Threads REQUIRED)
file(GLOB sdk_sources CONFIGURE_DEPENDS "${api}/*.cpp" "${api}/protobufUnix/*.cc")
add_library(twsapi STATIC ${sdk_sources})
target_compile_features(twsapi PUBLIC cxx_std_17)
set_target_properties(twsapi PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(twsapi SYSTEM PUBLIC "${api}" "${api}/protobufUnix")
target_link_libraries(twsapi PUBLIC protobuf::libprotobuf Threads::Threads "${IBKR_DECIMAL_LIBRARY}")
